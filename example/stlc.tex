\begin{inference}[rule_set=STLC syntax]
──────────────────────────── S-Base
ι type

A type
B type
──────────────────────────── S-Arrow
A → B type

──────────────────────────── S-Var
x term

A type
e term
──────────────────────────── S-Abs
λ x : A . e term

e₁ term
e₂ term
──────────────────────────── S-App
e₁ e₂ term

──────────────────────────── S-Empty
∅ ctx

Γ ctx
A type
x ∉ dom Γ
──────────────────────────── S-Extend
Γ, x : A ctx
\end{inference}

\begin{inference}[rule_set=STLC typing]
x : A ∈ Γ
──────────────────────────── T-Var
Γ ⊢ x : A

x ∉ dom Γ
Γ, x : A ⊢ e : B
──────────────────────────── T-Abs
Γ ⊢ λ x : A . e : A → B

Γ ⊢ e₁ : A → B
Γ ⊢ e₂ : A
──────────────────────────── T-App
Γ ⊢ e₁ e₂ : B
\end{inference}
